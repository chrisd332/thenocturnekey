        // Switch between Property Exterior and Grand Suite views without reload
        function toggleCardView(hotelId, mode) {
            cardViewModes[hotelId] = mode;
            const hotel = hotelsData.find(h => h.id === hotelId);
            if (!hotel) return;

            const imgEl = document.getElementById(`img-${hotelId}`);
            if (imgEl) {
                imgEl.style.opacity = '0.3';
                setTimeout(() => {
                    imgEl.src = mode === 'exterior' ? hotel.exteriorImage : hotel.suiteImage;
                    imgEl.style.opacity = '1';
                }, 150);
            }
            renderHotelCards();
        }

        // Filter by City via Header and Navigation
        function setCityFilter(city) {
            currentCityFilter = city;
            
            // Sync Dropdown
            const selectEl = document.getElementById('filterCity');
            if (selectEl) selectEl.value = city;

            // Update Nav Pill Styles
            document.querySelectorAll('.city-nav-link').forEach(btn => {
                if (btn.dataset.city === city) {
                    btn.classList.add('text-gold', 'bg-gold/10', 'border-gold/30');
                    btn.classList.remove('text-pearl', 'border-transparent');
                } else {
                    btn.classList.remove('text-gold', 'bg-gold/10', 'border-gold/30');
                    btn.classList.add('text-pearl', 'border-transparent');
                }
            });

            // Update Heading Label
            const heading = document.getElementById('activeFilterHeading');
            if (city === 'all') {
                heading.innerText = 'The Sovereign Global Index';
            } else {
                const formatted = city.replace('-', ' ').toUpperCase();
                heading.innerText = `Top Sanctuaries in ${formatted}`;
            }

            renderHotelCards();
        }

        // Filter by Atmosphere/Vibe tag
        function quickFilterTag(vibe) {
            currentVibeFilter = vibe;
            const vibeSelect = document.getElementById('filterVibe');
            if (vibeSelect) vibeSelect.value = vibe;

            document.querySelectorAll('.vibe-tag').forEach(tag => {
                if (tag.innerText.toLowerCase().includes(vibe.toLowerCase()) || (vibe === 'all' && tag.innerText === 'ALL VIBES')) {
                    tag.classList.add('border-gold/40', 'bg-gold/10', 'text-gold-light');
                    tag.classList.remove('border-borderMuted', 'bg-charcoal/50', 'text-pearl');
                } else {
                    tag.classList.remove('border-gold/40', 'bg-gold/10', 'text-gold-light');
                    tag.classList.add('border-borderMuted', 'bg-charcoal/50', 'text-pearl');
                }
            });

            renderHotelCards();
        }

        // Dropdown bar trigger
        function handleDropdownFilter() {
            const cityVal = document.getElementById('filterCity').value;
            const vibeVal = document.getElementById('filterVibe').value;
            currentCityFilter = cityVal;
            currentVibeFilter = vibeVal;
            setCityFilter(cityVal);
            quickFilterTag(vibeVal);
        }

        function scrollToResults() {
            document.getElementById('listingsSection').scrollIntoView({ behavior: 'smooth' });
        }

        function toggleMobileNav() {
            const drawer = document.getElementById('mobileDrawer');
            drawer.classList.toggle('hidden');
        }

        document.getElementById('mobileMenuBtn').addEventListener('click', toggleMobileNav);

        // Share Concierge Link Handler
        function copyPropertyShare(hotelName, url) {
            const dummy = document.createElement('input');
            document.body.appendChild(dummy);
            dummy.value = `${hotelName} on The Nocturne Key: ${url}`;
            dummy.select();
            document.execCommand('copy');
            document.body.removeChild(dummy);

            // Subtle Toast
            const toast = document.createElement('div');
            toast.className = 'fixed bottom-6 right-6 z-50 bg-gold text-obsidian px-4 py-2.5 rounded-xl font-mono text-xs uppercase tracking-widest shadow-2xl transition-soft';
            toast.innerText = 'Sanctuary Link Copied to Clipboard';
            document.body.appendChild(toast);
            setTimeout(() => {
                toast.style.opacity = '0';
                setTimeout(() => toast.remove(), 400);
            }, 2500);
        }

        // Newsletter submission handler
        function handleSubscribe(e) {
            e.preventDefault();
            const emailInput = document.getElementById('subscriberEmail');
            const feedback = document.getElementById('subscribeFeedback');
            
            if (emailInput.value) {
                feedback.innerText = `Key granted. Welcome to the circle, ${emailInput.value}. Watch your inbox at midnight.`;
                feedback.classList.remove('hidden');
                emailInput.value = '';
            }
        }

        // Initial Mount
        window.addEventListener('DOMContentLoaded', () => {
            renderHotelCards();
        });
