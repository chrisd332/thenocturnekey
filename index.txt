    <script>
        // Dual image view switcher (Property vs Suite)
        function swapImage(imgId, newSrc, btnElement) {
            const targetImg = document.getElementById(imgId);
            if (!targetImg) return;
            
            targetImg.style.opacity = '0.3';
            setTimeout(() => {
                targetImg.src = newSrc;
                targetImg.style.opacity = '1';
            }, 150);

            const parentButtons = btnElement.parentElement.querySelectorAll('.view-btn');
            parentButtons.forEach(btn => {
                btn.className = "view-btn flex-1 py-1.5 text-[11px] uppercase tracking-wider rounded bg-obsidian/60 text-pearl hover:text-ivory backdrop-blur-sm transition-soft";
            });
            btnElement.className = "view-btn flex-1 py-1.5 text-[11px] uppercase tracking-wider rounded bg-obsidian/85 text-gold border border-gold/40 backdrop-blur-sm transition-soft";
        }

        // City filtering logic
        function selectCityAndOpenHotels(cityName) {
            const cards = document.querySelectorAll('.hotel-card');
            const buttons = document.querySelectorAll('.city-filter-btn');
            const titleElem = document.getElementById('currentCategoryTitle');
            const countElem = document.getElementById('hotelCount');
            
            let count = 0;

            cards.forEach(card => {
                const cardCity = card.getAttribute('data-city');
                if (cityName === 'all' || cardCity === cityName) {
                    card.style.display = 'block';
                    count++;
                } else {
                    card.style.display = 'none';
                }
            });

            // Update title and count pill
            if (titleElem) {
                titleElem.textContent = cityName === 'all' ? 'Global Capitols Showcase' : `${cityName.replace('-', ' ')} Curated Guide`;
            }
            if (countElem) {
                countElem.textContent = `${count} ${count === 1 ? 'Property' : 'Properties'} Listed`;
            }

            // Update active state of buttons
            buttons.forEach(btn => {
                if (btn.getAttribute('data-city') === cityName) {
                    btn.className = "city-filter-btn text-xs uppercase tracking-widest px-2.5 py-1.5 rounded-full text-gold border border-gold/30 bg-gold/5 transition-soft";
                } else {
                    btn.className = "city-filter-btn text-xs uppercase tracking-widest px-2.5 py-1.5 rounded-full text-pearl hover:text-ivory border border-transparent hover:border-borderMuted transition-soft";
                }
            });
        }

        // Mobile drawer toggle
        const mobileMenuBtn = document.getElementById('mobileMenuBtn');
        const mobileDrawer = document.getElementById('mobileDrawer');
        
        function toggleMobileNav() {
            if (mobileDrawer) mobileDrawer.classList.toggle('hidden');
        }

        if (mobileMenuBtn) {
            mobileMenuBtn.addEventListener('click', toggleMobileNav);
        }

        // Detect ?city= parameter from capitols.html and auto-scroll to the hotel feed
        window.addEventListener('DOMContentLoaded', () => {
            const params = new URLSearchParams(window.location.search);
            const cityParam = params.get('city');

            if (cityParam) {
                selectCityAndOpenHotels(cityParam);
                const targetFeed = document.getElementById('hotel-feed');
                if (targetFeed) {
                    setTimeout(() => {
                        targetFeed.scrollIntoView({ behavior: 'smooth' });
                    }, 200);
                }
            }
        });
    </script>
