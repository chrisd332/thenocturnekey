  <script>
    function switchMedia(imageId, viewType, clickedBtn) {
      const img = document.getElementById(imageId);
      if (!img) return;

      // Update button visual state
      const buttonGroup = clickedBtn.parentElement;
      buttonGroup.querySelectorAll('.toggle-btn').forEach(btn => btn.classList.remove('active'));
      clickedBtn.classList.add('active');

      // Smooth image transition
      img.style.opacity = '0.35';
      setTimeout(() => {
        if (viewType === 'property') {
          img.src = img.getAttribute('data-property');
        } else if (viewType === 'suite') {
          img.src = img.getAttribute('data-suite');
        }
        img.style.opacity = '1';
      }, 160);
    }
  </script>
